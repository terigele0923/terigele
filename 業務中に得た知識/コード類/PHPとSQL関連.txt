<?php
function getPDO(): PDO {
    $host     = 'localhost';
    $dbname   = 'your_database_name';
    $username = 'your_username';
    $password = 'your_password';
    $charset  = 'utf8mb4';

    $dsn = "mysql:host=$host;dbname=$dbname;charset=$charset";

    try {
        $pdo = new PDO($dsn, $username, $password, [
            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION, // 例外でエラー通知
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,       // 結果を連想配列で
            PDO::ATTR_EMULATE_PREPARES   => false,                  // ネイティブプリペアド
        ]);
        return $pdo;
    } catch (PDOException $e) {
        // 本番環境ではログに出力して、画面には一般的なエラー文のみ表示推奨
        die("データベース接続に失敗しました: " . htmlspecialchars($e->getMessage()));
    }
}
?>


function getPDO():PDO{
	$host = '';
	$dbname = '';
	$username = '';
	$password = '';
	$charset = '';

	$dsn = "mysql:host=$host;dbname=$dbname;charset=$charset";

	try{
		$pdo = new PDO($dsn,$username,$password,[
			PDO::ATTR_ERRMODE	=> PDO::ERRMODE_EXCEPTION,
			PDO::ATTR_DEFAULT_FETCH_MODE	=> PDO::FETCH_ASSOC,
			PDO::ATTR_EMULATE_PREPARES	=>false,
		]);
		return $pdo;
	}catch(PDOException $e){
		die("データベース接続に失敗しました：".htmlspecialchars($e->getMessage()));
	}
}

isset() の意味
「変数が定義されていて、かつ null ではないか」**をチェックします。
isset($var) は「$var が存在していて null ではない」ときに true を返します。

empty() の意味
**「値が空かどうか（0, '', null, false, [], '0' など）」**をチェックします。

empty($var) は「$var が未定義、もしくは空っぽ」なら true を返します。
