Git　コマンド(サーバー上のファイルを取り戻す)

cd .../app/dir/
# ステージングしていない全てのファイルをリセット
git checkout . 

# test.phpのみ
git checkout ./test.php

# ローカルのコミットをハードリセット
git reset --hard 戻りたい時点のsha1(ハッシュ値)

# ローカルのハードリセットしたステータスをリモートに反映
git push -f origin 戻りたい時点のsha1:ブランチ名

# ローカルのステータスを強制的にoriginに揃える
git reset --hard origin/release/yougo-ai.com

# appディレクトリの権限を開放（一時的に）
chmod -R 777 /export/www/hosting/bis-hop.jp/yosou002/p-system.biz
chown -R permadmin:permadmin /export/www/hosting/bis-hop.jp/yosou002/p-system.biz/

ssh -p 22 -o HostKeyAlgorithms=+ssh-rsa 接続ユーザー@ホスト
cat << 'EOF' >> ~/.ssh/config
Host p-system.biz
  User permadmin
  HostName 103.204.64.20
  Port 22
  HostKeyAlgorithms +ssh-rsa
  PubkeyAcceptedKeyTypes +ssh-rsa

Host p-system.biz_db
  User permadmin
  HostName 103.204.64.21
  Port 22
  HostKeyAlgorithms +ssh-rsa
  PubkeyAcceptedKeyTypes +ssh-rsa
EOF
